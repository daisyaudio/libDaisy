82 Result Open(const char* name)
83 {
84 if(is_open_)
85 f_close(&file_);
86 auto sta = f_open(&file_, name, (FA_OPEN_EXISTING | FA_READ));
87 switch(sta)
88 {
89 case FR_OK: break;
90 case FR_NO_FILE:
91 case FR_NO_PATH: return Result::FileNotFoundError;
92 default: return Result::DiskError;
93 }
94 is_open_ = true;
95
96 daisy::FileReader reader(&file_);
97 daisy::WavParser parser;
98 if(!parser.parse(reader))
99 return Result::DiskError;
100 const auto& info = parser.info();
101 file_info_.channels = info.numChannels;
102 file_info_.samplerate = info.sampleRate;
103 auto bd = info.bitsPerSample;
104 file_info_.data_size_bytes = parser.dataSize();
105 file_info_.length
106 = file_info_.data_size_bytes / ((bd / 8) * file_info_.channels);
107 file_info_.data_start = parser.dataOffset();
108
109 // Compute frame size (bytes per sample frame)
110 frame_bytes_ = (file_info_.channels) * (bd / 8);
111 if(frame_bytes_ == 0)
112 return Result::DiskError;
113
114 // Seek to start of data
115 if(f_lseek(&file_, file_info_.data_start) != FR_OK)
116 return Result::DiskError;
117
118 // Prime FIFO with frame-aligned read
119 std::fill(buff_raw_, buff_raw_ + kRxSizeSamples, 0);
120 UINT bytes_read = 0;
121 size_t bytes_to_read
122 = std::min((size_t)workspace_bytes, file_info_.data_size_bytes);
123 // Align down to full frames
124 bytes_to_read -= (bytes_to_read % frame_bytes_);
125 if(bytes_to_read > 0)
126 {
127 if(f_read(&file_, (void*)buff_raw_, bytes_to_read, &bytes_read)
128 != FR_OK)
129 {
130 f_close(&file_);
131 return Result::DiskError;
132 }
133 if(bytes_read != bytes_to_read)
134 {
135 f_close(&file_);
136 return Result::DiskError;
137 }
138 }
139
140 buff_fifo_.Clear();
141 size_t samps_to_write = bytes_read / sizeof(int16_t);
142 // Align push count to whole frames (multiples of channels)
143 if(file_info_.channels > 0)
144 samps_to_write -= (samps_to_write % file_info_.channels);
145 for(size_t i = 0; i < samps_to_write; i++)
146 {
147 if(!buff_fifo_.PushBack(buff_raw_[i]))
148 break;
149 }
150
151 position_ = 0;
152 pending_read_req_ = false;
153 awaiting_first_data_ = false;
154 pending_seek_req_ = false;
155 handled_gen_ = restart_gen_;
156 bytes_left_in_chunk_ = (bytes_read <= file_info_.data_size_bytes)
157 ? (file_info_.data_size_bytes - bytes_read)
158 : 0;
159
160 return Result::Ok;
161 }

181 Result Prepare()
182 {
183 for(;;)
184 {
185 // Capture the restart generation, used to discard unwanted data
186 // if a restart occurs in the callback during Disk I/O.
187 const uint32_t gen = restart_gen_;
188 if(gen != handled_gen_)
189 {
190 handled_gen_ = gen;
191 request_fifo_.Clear();
192 if(f_lseek(&file_, file_info_.data_start) != FR_OK)
193 {
194 Close();
195 return Result::DiskError;
196 }
197 bytes_left_in_chunk_ = file_info_.data_size_bytes;
198 // Smaller first read gets audio started sooner (less underrun)
199 size_t first = kRxSizeSamples / 4;
200 auto res = ReadIntoFifo(first, gen);
201 if(res != Result::Ok)
202 {
203 if(res == Result::DiskError)
204 Close();
205 return res;
206 }
207 continue;
208 }
209 if(request_fifo_.IsEmpty())
210 break;
211 auto req = request_fifo_.PopFront();
212 if(req.type != IoRequest::Type::Read)
213 continue;
214 auto res = ReadIntoFifo(req.data, gen);
215 if(res != Result::Ok)
216 {
217 if(res == Result::DiskError)
218 Close();
219 return res;
220 }
221 }
222 return Result::Ok;
223 }

249 Result Stream(float* samples, size_t num_channels)
250 {
251 auto channels = file_info_.channels;
252
253 for(size_t i = 0; i < num_channels; i++)
254 samples[i] = 0.f;
255
256 // immediately return if no file is open.
257 // could probably use a different result name that's more specific
258 if(!is_open_)
259 return Result::FileNotFoundError;
260
261 if(!buff_fifo_.IsEmpty() && playing_)
262 {
263 awaiting_first_data_ = false;
264 size_t ch_out = std::min(channels, num_channels);
265 for(size_t i = 0; i < ch_out; i++)
266 {
267 samples[i]
268 = previous_sample_[i]
269 + pos_acc_ * (current_sample_[i] - previous_sample_[i]);
270 }
271
272 pos_acc_ += playback_speed_;
273 while(pos_acc_ >= 1.f)
274 {
275 position_ += 1;
276 pos_acc_ -= 1.f;
277 for(size_t i = 0; i < channels; i++)
278 {
279 previous_sample_[i] = current_sample_[i];
280 current_sample_[i] = s162f(buff_fifo_.PopFront());
281 }
282 }
283 }
284
285 if(position_ >= (file_info_.length > 0 ? file_info_.length : 1))
286 {
287 position_ = 0;
288 if(!looping_)
289 {
290 playing_ = false;
291 }
292 else
293 {
294 pos_acc_ = 0.f;
295 for(size_t i = 0; i < kMaxAudioChannels; i++)
296 previous_sample_[i] = current_sample_[i];
297 }
298 }
299
300 // Request new samples in whole frames
301 bool requested_new_samps = false;
302 if(buff_fifo_.GetNumElements() < kRxFifoThreshold && !pending_read_req_)
303 {
304 size_t free_slots = (kRxSizeSamples - buff_fifo_.GetNumElements());
305 size_t rx_qty = (free_slots > 1) ? (free_slots - 1) : 0;
306 // Align to multiples of channels
307 if(file_info_.channels > 0)
308 rx_qty -= (rx_qty % file_info_.channels);
309
310 if(rx_qty > 0)
311 {
312 request_fifo_.PushBack(
313 IoRequest(IoRequest::Type::Read, rx_qty));
314 pending_read_req_ = true;
315 requested_new_samps = true;
316 }
317 }
318 if(requested_new_samps)
319 return Result::NewSamplesRequested;
320 else if(buff_fifo_.IsEmpty() && playing_)
321 return awaiting_first_data_ ? Result::InitialLatency
322 : Result::PlaybackUnderrun;
323 else
324 return Result::Ok;
325 }

328 void Restart()
329 {
330 pos_acc_ = 0.f;
331 for(size_t i = 0; i < kMaxAudioChannels; i++)
332 current_sample_[i] = previous_sample_[i] = 0.f;
333 if(!is_open_)
334 return;
335 pending_read_req_ = true;
336 awaiting_first_data_ = true;
337 position_ = 0;
338 playing_ = true;
339 buff_fifo_.Clear();
340 // Restart generation increments will flag the `Prepare()` method to
341 // discard any stale data that is in progress.
342 restart_gen_ = restart_gen_ + 1;
343 }

412 Result ReadIntoFifo(size_t samples, uint32_t gen)
413 {
414 size_t bytes_requested = samples * sizeof(int16_t);
415 // Align to full frames
416 bytes_requested -= (bytes_requested % frame_bytes_);
417 if(bytes_requested == 0)
418 {
419 if(gen == restart_gen_)
420 pending_read_req_ = false;
421 return Result::Ok;
422 }
423
424 UINT total_bytes_read = 0;
425 UINT bytes_read = 0;
426
427 std::fill(buff_raw_, buff_raw_ + kRxSizeSamples, 0);
428
429 // Read up to end of data chunk
430 size_t first_span = std::min(bytes_requested, bytes_left_in_chunk_);
431 // Align first_span to frame boundary as well
432 first_span -= (first_span % frame_bytes_);
433 if(first_span > 0)
434 {
435 if(f_read(&file_, (void*)buff_raw_, first_span, &bytes_read)
436 != FR_OK)
437 return Result::DiskError;
438 if(bytes_read != first_span)
439 return Result::DiskError;
440 total_bytes_read += bytes_read;
441 bytes_left_in_chunk_ -= bytes_read;
442 }
443
444 // If need more and looping, wrap and read more (frame-aligned)
445 if(total_bytes_read < bytes_requested && looping_)
446 {
447 if(f_lseek(&file_, file_info_.data_start) != FR_OK)
448 return Result::DiskError;
449
450 size_t remaining_bytes = bytes_requested - total_bytes_read;
451 // Align remaining as well (it already is, but keep consistent)
452 remaining_bytes -= (remaining_bytes % frame_bytes_);
453 if(remaining_bytes > 0)
454 {
455 UINT bytes_read2 = 0;
456 char* tbuff = ((char*)(buff_raw_) + total_bytes_read);
457 // Do not exceed the chunk size on wrap
458 size_t span2
459 = std::min(remaining_bytes, file_info_.data_size_bytes);
460 // Align span2
461 span2 -= (span2 % frame_bytes_);
462 if(span2 > 0)
463 {
464 if(f_read(&file_, (void*)tbuff, span2, &bytes_read2)
465 != FR_OK)
466 return Result::DiskError;
467 if(bytes_read2 != span2)
468 return Result::DiskError;
469 total_bytes_read += bytes_read2;
470 bytes_left_in_chunk_
471 = (bytes_read2 <= file_info_.data_size_bytes)
472 ? (file_info_.data_size_bytes - bytes_read2)
473 : 0;
474 }
475 }
476 }
477
478 // Count total samples to write to FIFO
479 size_t samps_to_write = total_bytes_read / sizeof(int16_t);
480 if(file_info_.channels > 0)
481 samps_to_write -= (samps_to_write % file_info_.channels);
482
483 // Push samples to FIFO in chunks, disabling IRQs during each
484 // chunk. A Restart (from the audio callback) will cause the
485 // remaining data to be discarded to avoid unwanted, stale data.
486 constexpr size_t kPushChunk = 128;
487 size_t i = 0;
488 while(i < samps_to_write)
489 {
490 size_t end = std::min(i + kPushChunk, samps_to_write);
491 uint32_t primask = __get_PRIMASK();
492 __disable_irq();
493 if(gen != restart_gen_)
494 {
495 __set_PRIMASK(primask);
496 return Result::Ok; // stale: discard
497 }
498 bool full = false;
499 for(; i < end; i++)
500 {
501 if(!buff_fifo_.PushBack(buff_raw_[i]))
502 {
503 full = true;
504 break;
505 }
506 }
507 __set_PRIMASK(primask);
508 if(full)
509 {
510 pending_read_req_ = false;
511 return Result::PrepareOverrun;
512 }
513 }
514 pending_read_req_ = false;
515 return Result::Ok;
516 }
